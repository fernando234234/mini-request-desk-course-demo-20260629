    $path = $normalize($path);
    $prefix = rtrim($normalize($root), '/') . '/';
    return strpos($path, $prefix) === 0 ? substr($path, strlen($prefix)) : basename($path);

$parseConfigLiteral = function ($source, $name) {
    $pattern = '/define\\s*\\(\\s*([\'\"])' . preg_quote($name, '/') . '\\1\\s*,\\s*([\'\"])(.*?)\\2\\s*\\)/s';
    $count = preg_match_all($pattern, $source, $matches, PREG_SET_ORDER);
    if ($count !== 1 || strpos($matches[0][3], '\\') !== false || strpos($matches[0][3], '$') !== false) {
        return null;
    }
    return $matches[0][3];
};
